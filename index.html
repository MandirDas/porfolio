03 / Approach

How I engineer apps

Clear layers, defensive data handling, and a release pipeline that does the repetitive work.

sip_bloc.dart illustrative 
sealed class SipEvent {}
final class SipSubmitted extends SipEvent {
  SipSubmitted(this.amount);
  final Money amount;
}

class SipBloc extends Bloc<SipEvent, SipState> {
  SipBloc(this._createSip) : super(const SipState.idle()) {
    on<SipSubmitted>(_onSubmitted);
  }

  final CreateSip _createSip; // domain use case

  Future<void> _onSubmitted(
    SipSubmitted e, Emitter<SipState> emit) async {
    emit(const SipState.submitting());
    final result = await _createSip(e.amount);
    emit(result.fold(SipState.failure, SipState.success));
  }
}
